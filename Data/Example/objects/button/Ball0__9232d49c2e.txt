local myName,W,du,dv

on mouseDown
  if the optionKey is down then
    grab me
    exit mouseDown
  end if
  
  put the width of this card into W
  put the name of me into myName
end mouseDown

on mouseUP
  if myName is "" then exit mouseUp
  set the points of grc "v" to ""
  set the points of grc "vExtension" to ""
  repeat with i = 1 to 15
    put "scatCross" & i into tName
    set the visible of grc tName to false
  end repeat
  startStroke du/25,dv/25
  put "" into myName
end mouseUP

on mouseLeave
  mouseUP
end mouseLeave
