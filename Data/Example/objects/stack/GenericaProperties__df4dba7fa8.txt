on writeOutlineToFiles tFolderPath
   set itemDel to comma
   if tFolderpath is not empty then put tFolderPath into gWriteToDirectoryRoot
   put the customPropertySets of this card into gCardPropertySets
   filter gCardPropertySets without "cREV"&"*"
   repeat for each line L in gCardPropertySets
      -- focus the property set
      do "set the customPropertySet of this card to" && L
      
      -- truncate the name for folders
      replace "game_" with empty in L
      -- create a folder for the data
      put gWriteToDirectoryRoot & "/" & L into gCurrentFolderPath
      if there is not a folder gcurrentFolderPath then create folder gCurrentFolderPath
      
      -- the keys of the property set
      put the customKeys of this card into tFilesArray
      
      --iterate over the keys
      repeat for each line K in tFilesArray
         -- collect the data held in the key
         do "put the" && K &&"of this card into tArray"
         put the keys of tArray into tArrayKeys
         
         filter lines of tArrayKeys without "description"
         sort lines of tArraykeys numeric ascending
         replace comma with cr in tArraykeys
         put tab & k & cr after allKeys
         put "" into tArrayTextContent
         -- compile the data to a single variable
         repeat for each line tIndex in tArrayKeys
            put   tIndex & cr after allKeys
            put tArray[tIndex] into tCleanText
            replace "[" with empty in tCleanText
            replace "]" with empty in tCleanText
            set itemDel to "."
            put item 2 to -1 of tCleanText into tCleanText
            put tCleanText into tArray[tIndex]
            put tCleanText & cr after tArrayTextContent
            set the k of this card to tarray
         end repeat
         filter lines of tArrayTextContent without empty
 
         put tArrayTextContent into tArray[tIndex]
         -- write data to file
         put "file:" & gCurrentFolderPath &"/" & K &".txt" into gCurrentFilePath
         put tArrayTextContent into URL gCurrentFilePath
         
         put "file:" & gCurrentFolderPath &"/" & K &"_description.txt" into gCurrentFilePath
         put tArray["description"] into URL gCurrentFilePath
         put k && tArray["description"] & cr after allDescriptions
      end repeat
   end repeat
   put allDescriptions into URL ("file:" & gWriteToDirectoryRoot &"/" & "property_descriptions.txt")
   -- UI_Properties is an extra level of detail
   writeUIPropertiesArraysToFiles
end WriteOutlineToFiles

Function sortNumericWorkAround tListOfNumbers
   put 0 into tLastNumber
   
   repeat while number of lines of tListOfNumbers > 0
      put number of lines of tListOfNumbers into i
      put line i of tListOfNumbers into tCheck
      if n+1 > number of lines of tListofNumbers then put 0 into N
      add 1 to n
      put line n of tList into tCheck2
      if tCheck > tLastNumber then 
         put L &cr  after tNewList
         put L into tLastNumber
         next repeat
      end if
   end repeat
   return tNewList
end SortNumericWorkAround

      repeat for each line tIgnore in gGlobalPrefixIgnoreList
         filter lines of gGlobalList without tIgnore& "*"
      end repeat
   end if
   
   sort gGlobalList
   repeat for each line L in gGlobalList
      do "put" && L &&"into tVar"
      try
         do "put L &&" && quoted(tVar)  &&"&& cr after tAllCurrentGlobalValues"
      catch err
         put err & cr after tAllCurrentGlobalValues
      finally

on ImportGamePropertyArrays
   set the customPropertySets of this card to empty
   
   set the customPropertySet of this card to NewTemplate
   set the customKeys of this card to empty
   
   -- iterate over the folders of directory 
   repeat for each line gFirstLevelFolder in gSourceDirectoryFolders
      put gSourceDirectoryRoot & "/" & gFirstLevelFolder into gFirstLevelFolderPath
      
      -- a prefix needs to be added to avoid interfering with the IDE properties
      put "game_"&gFirstLevelFolder into gFirstLevelFolderArrayLabel
      do "set the customPropertySet of this card to" &&  gFirstLevelFolderArrayLabel
      
      set the defaultFolder to gFirstLevelFolderPath
      put gFirstLevelFolderPath into field "pathfield"
      put the files into gCurrentFileList
      --isolate the description text from the label lists
      put gCurrentFileList into gLabelsFilesList
      --filter lines of gLabelsFilesList with "*"&"labels"&"*"
      filter lines of gLabelsFilesList without "*"&"_description"&"*"
      put gCurrentFileList into gLabelsDescription
      filter lines of gLabelsDescription with "*"&"_description"&"*"
      if gLabelsDescription is not empty then put gLabelsDescription&  cr after allDescri
      -- for now we only have these label files, what if more ?
      put gLabelsFilesList into gCurrentFileList
      filter lines of gCurrentFileList without empty
      --- iterate over the files of the first level of folders
      repeat for each line gFirstLevelFile in gCurrentFileList
         --- get array named after the folder
         
         --- make sure to focus the property set 
         do "set the customPropertySet of this card to" &&  gFirstLevelFolderArrayLabel
         
         --- get the content of the file
         put "file:"& gFirstLevelFolderPath & "/" & gFirstLevelFile into gFirstLevelFilePath
         put gFirstLevelFilePath into field "pathfield"
         put URL  gFirstLevelFilePath  into gFirstLevelText
         
         --- remove extension
         put char 1 to -5 of gFirstLevelFile into gFirstLevelArrayName
         
         --- remove "game_" header
         set ItemDel to "_"
         put gFirstLevelArrayName into gFirstLevelSubArrayLabel
         
         --- iterate over the text
         put 0 into N
         filter lines of gFirstLevelText without empty
         repeat for each line L in gFirstLevelText
            add 1 to N
            --- move each line of text to an array  index
            put L into gFirstLevelSubArrayLabel[padNumber(N,3)]
         end repeat
         
         -- import the the description data and insert it amng the things here.
         if gLabelsDescription is not empty then
            put textfile.data(gFirstLevelFolderPath,gLabelsDescription) into gLabelsDescriptionText
            put gLabelsDescriptionText into gFirstLevelSubArrayLabel["description"]
            
            -- lets try something special
            set the customPropertySet of this card to description
            put the customKeys of this card into tKeyList
            put the game of this card into gGameDescriptionsArray
            --put padNumber(number of lines of tKeyList+1,3) into tNewKey
            replace "_description.txt" with empty in gLabelsDescription
            replace "game_" with empty in gLabelsDescription
            put gLabelsDescriptionText into gGameDescriptionsArray[gLabelsDescription]
            set the game of this card to gGameDescriptionsArray
            do "set the customPropertySet of this card to" &&  gFirstLevelFolderArrayLabel
         end if
         
         -- store the array data in the custom property set of the card
         if gFirstLevelArrayName is not empty and gFirstLevelSubArrayLabel is not empty then
            do "set the" && gFirstLevelArrayName &&"of this card to gFirstLevelSubArrayLabel"
         end if
      end repeat
   end repeat
   put allDescri
   set the customPropertySet of this card to empty
end ImportGamePropertyArrays

function padNumber tNum,tLen
   put "padNumber("& tNum,tLen & ")" & cr after gDebug
   put number of characters of tNum into tCount
   repeat with i = 1 to tLen-tCount
      put 0 before tNum
   end repeat
   return tNum
end padNumber

function filterStackScriptForGlobals tGlobalPrefix
   if tGlobalPrefix is an integer then put numtochar(tGlobalPrefix) into tGlobalPrefix
   if tGlobalPrefix is  empty then put "g" into tGlobalPrefix
   put tGlobalPrefix into gGlobalPrefixChar
   put the script of this stack into tScript
   put tScript into tGlobals
   filter tGlobals with "global " &"*"
   --filter tscript without "global " &"*"
   repeat with i = 1 to 64
      if  i < 10 or i > 11  or i <> 46 or i <> 95 then
         replace numtochar(i) with cr in tScript
      end if
   end repeat
   filter lines of tScript without empty
   repeat with i = 91 to 255
      if i < 97 or i > 122 and i <> 46 and i <> 95 then
         replace  numtochar(i) with cr in tScript
      end if
   end repeat
   filter lines of tScript with "*"&gGlobalPrefixChar&"*"
   filter lines of tScript with gGlobalPrefixChar&"*"
   put "" into gFoundGlobals
   repeat with i = number of lines of tScript down to 1
      put line i of tscript into tGlobalName
      filter  lines of tScript  without tGlobalName
      -- only matching character for globals gGlob DO_SOMETHING etc
      if chartonum(char 1 of tGlobalName) = chartonum(char 1 of gGlobalPrefixChar) then
         if chartonum(char 2 of tGlobalName) >= 65 and chartonum(char 2 of tGlobalName) <=  90 then 
            put tGlobalName & cr after gFoundGlobals
         end if
      end if
   end repeat
   filter lines of gFoundGlobals without empty
   sort gFoundGlobals
   return gFoundGlobals
end filterStackScriptForGlobals

on objectPropertyDimensionChanged tFocusedField
   put the customPropertySet of this card into gCurrentPropertySet 
   set the customPropertyset of this card to Game_UI_Properties
   put "width,height,x,y,left,top,right,bottom" into gButtonPropertyGeometryFields
   set itemDel to comma
   lock screen
   lock messages
   repeat for each item I in gButtonPropertyGeometryFields
      do "put field"&& quote & I&   "field" &quote&&"into g" &I
   end repeat
   switch tFocusedField
      case "widthField"
         put gLeft+gWidth into gRight
         put gRight into field "rightField"
         put gWidth/2 into gX
         put gX into field "xField"
         break
      case "heightField"
         put gTop+gHeight into gBottom
         put gBottom into field "bottomField"
         put gHeight/2 into gY
         put gY into field "xField"
         break
         
      case "leftField"
      case "rightField"
         put  gRight - gLeft  into gWidth
         put gWidth into field "wdithfield"
         put gLeft+gWidth /2 into gX
         put gX into field "xField"
         break
         
      case "topField"
      case "bottomField"
         put gTop - gBottom into gHeight
         put gHeight into field "heightfield"
         put gTop+gHeight /2 into gY
         put gY into field "yField"
         break
   end switch
   unlock messages
   unlock screen
   updateUIPropertiesDataFromControls
   set the customPropertySet of this card to gCurrentPropertySet
end objectPropertyDimensionChanged

on SCRIPT_CARD_BuildGlobalInit
   put cr & the script of this stack into tScript
   filter tScript with "global "&"*"
   replace  "global" with empty in tScript
   repeat with i = 97 to 122
      put filterStackScriptForGlobals(numtochar(i)) & cr after  gGlobalList
   end repeat
   filter lines of gGlobalList without empty
   set the customPropertySet of this card to empty
   put the global_prefix_ignore_list of this card into gGlobalPrefixIgnoreList
   if gGlobalPrefixIgnoreList is not empty then
      repeat for each line tIgnore in gGlobalPrefixIgnoreList
         filter lines of gGlobalList without tIgnore& "*"
      end repeat
   end if
   
   put "o"&"n"&&"init"&"NewGlobals" & cr into gGlobalInitBuild  
   sort gGlobalList
   repeat for each line L in gGlobalList
      filter gGlobalInitCatch without L
      put "" into gGlobalInitCatch
      switch (char 1 of l)
         case "b" 
            put "put false into" && L into  gGlobalInitCatch
            break
         case "i"
            put "put 1 into" && L into  gGlobalInitCatch
            break
         case "f"
            put "put 0 into" && L into  gGlobalInitCatch
            break
         case "s"
            put "put" && quote &L & quote &&"into" && L into gGlobalInitCatch
            break
      end switch
      
      if gGlobalInitCatch is empty then 
         put tab & "put" && quote & quote &&  "into" && L & cr after gGlobalInitBuild
      else
         put tab & gGlobalInitCatch& cr after gGlobalInitBuild
      end if
      put "global" && L & cr after gStackScriptGlobalHeader
   end repeat
   
   put "e"&"n"&"d init"&"NewGlobals" & cr & "---END OF STACK SCRIPT---" after gGlobalInitBuild
   
   
   -- find the init new globals handler at the end of the stack script
   put the script of this stack into tScript
   put  "---BEGIN_GLOBAL_INIT---" into gGlobalInitBuildHeader
   filter lines of tScript without  gGlobalInitBuildHeader
   put gGlobalInitBuildHeader & cr before gGlobalInitBuild
   put lineOffset("on"&&"init"&"NewGlobals",tscript)-1 into tInitHandlerLine
   if tInitHandlerLine <1 then 
      put cr after line -1 of tInitHandlerLine
      put -1 into tInitHandlerLine
   end if
   
   -- find the first handler or function in the stack script
   put lineoffset("on"&space,tScript) into tFirsthandlerLine
   put lineOffset("function",tScript) into tFirstFunctionLine
   if tFirsthandlerLine < tFirstFunctionLine then
      put tFirsthandlerLine into tScriptStartLineNumber
   else
      put tFirstFunctionLine into tScriptStartLineNumber
   end if
   -- combine the global header with the rest of the script
   put gStackScriptGlobalHeader & cr &  line tScriptStartLineNumber to tInitHandlerLine of tScript into tScriptWithHeader
   -- include the new global init handler at the end of the script
   put  tScriptWithHeader & cr & gGlobalInitBuild into tFinalStackScript
   
   -- save it to file in case things go bad
   put "file:"&specialFolderPath("Documents")&"/stack_script.livecodescript" into tFile
   put tFinalStackScript into URL tFile
   
   set the script of this stack to tFinalStackScript
end SCRIPT_CARD_BuildGlobalInit
