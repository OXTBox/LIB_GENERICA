      put L & cr after allLS

      put lineOffset("end"&&tTarget,tScript) into tEndPoint
      put line tStart to tEndPoint of tScript  &cr after tCopy
   end repeat
   --put number of lines of tCopy && SN
   set the clipboardData["text"] to tCopy
   set the script of the templateButton to tCopy 
   set the rect of the templateButton to 0,0,80,22
   put 22 into H
   set the top of the templateButton to top of group "Map"+(H*gNumberOfButtonRows)

      add 1 to gNumberOfButtonRows
   end if
   set the left of the templateButton to left of group "Map"+(gNumberOfButtons*width of button (short name of the target))
   set the name of the templateButton to "CLEANED OUT SCRIPT"
   set the opaque of the templateButton to false
   set the "default" of the templateButton to false

end ExtractStackScriptToButtons

function List.Globals tArg
   ops.setDebug ("00125 list.Globals" && tArg )
   put "" into gErrorIndent
   -- produces a clean list of the globals defined in the stack
   put the script of stack (the mainStack of this stack) into tScript
   put lineOffset("--"&"--ENDGLOBALDEFINITION--"&"--",tScript)-1 into ListStop
   put line 1 to ListStop of tScript into gGlobalsList
   filter lines of gGlobalsList without empty
   	   
   filter lines of gGlobalsList with "global"&"*"
   --replace "-"&"-" with space in
   --replace comma with cr in gGlobalsList 
   replace tab with space in gGlobalsList
   replace space&space with space in gGlobalsList
   replace "global" with empty gGlobalsList 
   	   
   repeat for each line L in gGlobalsList
      put word 1 of L & cr after tNewList
   end repeat
   put tNewList into gGlobalsList
   	   
   filter lines of gGlobalsList without empty
   if targ is "sort" then
      sort gGlobalsList
      repeat for each line L in gGlobalsList
         put "global" &tab& L & cr after tArgList
      end repeat
      return tArgList
   end if
   return gGlobalsList
end List.Globals

on ops.addErrorCodesToStackScript
   ops.setDebug ("00148 ops.addErrorCodesToStackScript")
   put "" into gErrorIndent
   set the cursor to busy
   put the script of this stack into tScript
   put tScript into tHands
   put tScript into tFuns
   filter lines of tHands with "on "&"*"
   sort tHands
   put 0 into tHandlerCount
   put "" into tModifiedStackScript
   repeat for each line tHand in tHands
      add 1 to tHandlerCount
      put "" into tHandlerBody
      put word 2 of tHand into tHandler
      	      
      put getHandlerFromStack(tHandler) into tHandlerBody
      filter lines of tHandlerBody without empty
      if the number of words of tHand > 2 then
         put "" into tArgs
         repeat for each item I in (word 3 to -1 of tHand)
            if char 1 of I is "-" then put ")" &I && "&&" after tArgs
            else
               put  I && "&&" after tArgs
            end if
         end repeat 
         put space & "&&" && char 1 to -3 of tArgs  after tErrorCodeLine
      end if
      put   line 1 of tHandlerBody & cr & tErrorCodeLine &")"  & cr into tHandlerUpate
      put  "put" && quote & quote && "into gErrorIndent" & cr & line 2 to -1 of tHandlerBody after tHandlerUpate
      if word 1 of line 1 of tHandlerUpate is numtochar(111)&numtochar(110)   then put tHandlerUpate & cr & cr after tModifiedStackScript
      if word 1 of line 1 of tHandlerUpate is "function"  then put tHandlerUpate & cr & cr after tModifiedStackScript
      string.writeFile "Handlers",word 2 of tHand,tHandlerUpate
   end repeat
   string.writeFile "StackScript","Handlers",tModifiedStackScript
   -- functions
   filter lines of tFuns with "function "&"*"
   sort tfuns
   repeat for each line tFunc in tFuns
      add 1 to tHandlerCount
      put "" into tHandlerBody
      put word 2 of tFunc into tFunctionLabel
      put getFunctionFromStack(tFunctionLabel) into tHandlerBody
      if the number of words of tHand > 2 then
         put "" into tArgs
         repeat for each item I in (word 3 to -1 of tHand)
            put  I && "&&" after tArgs
         end repeat 
         put space & "&&" && char 1 to -3 of tArgs  after tErrorCodeLine
      end if
      put  tErrorCodeLine &")" after char -1 of line 1 of tHandlerBody
      put cr &"put" && quote &quote & "into gErrorIndent" after line 2 of tHandlerBody
      replace space&space with empty in line 1 of   tHandlerBody
      put tHandlerBody & cr & cr after tModifiedStackScript
      string.writeFile "Functions",tFunctionLabel,tHandlerBody
   end repeat
   	   
   put tScript into tGlobals
   filter lines of tGlobals with "global"&"*"
   sort tGlobals
   replace tab with space in  tglobals
   replace space & space & space with space in tGlobals
   replace space & space & space with space in tGlobals
   replace space & space with space in tGlobals
   sort tGlobals
   put tGlobals & cr & "----END OF GLOBAL DEFINITIONS----"  into tGlobalsText
   put tGlobalsText & cr & cr before tModifiedStackScript
   string.writeFile "StackScript","StackScript_ErrorCodesAdded_"&string.asitems(the short date,"/","_"),tModifiedStackScript
   	   
   repeat for each line G in tGlobals
      put "put" && quote&quote &&"into" && G & cr after tBigInit
   end repeat
   	   
   put "on INIT.All_GLOBALS" & cr before tBigInit
   put  "end INIT.ALL_GLOBALS" after tBigInit
   put tGlobalsText & cr & cr before tBigInit
   string.writeFile "Globals","GlobalInit__"&string.asitems(the short date,"/","_"),tBigInit
   set the cursor to arrow
end ops.addErrorCodesToStackScript

on rawKeyDown tKey
   ops.setDebug ("00159 rawKeyDown" && tKey )
   put "" into gErrorIndent
   if the tool is not "browse tool" or   the mouseControl is not empty  then pass rawKeyDown
   set the label of button "theRawKey" to tkey & " = " & numtochar(tkey)
   	   
   	   
   switch tKey
      case 104 -- H key, center on selected
         UI.CenterOnSelected
         UI.ZoomedMap
         break
      case 97 -- "A KEY CREATES PATHS FOR CURRENT MOBILE GAME OBJECT"
         if word 1 of gSelectedObject is "mobile" then
            UI.TogglePathTool
            put gGameTool into field "statuslabel"
         end if
         break
      case 65288 -- "DELETE KEY CLEARS CURRENT WAYPOUT OF MOBILE GAME OBJECT."
         if gGameTool is "mobile" and gPathTooLBool then ClearSelectedPathPoint
         break
      case 65289 -- "TAB KEY  = ViSIT PLAYER GAME OBJECTS"
         put base.getPlayerBaseList(1) into gPlayerBaseList
         if number of lines of gPlayerBaseList > 1 then
            put lineOFfset( gGameObjectIndexLabel,gPlayerBaseList)+1 into tNext
         else
            put 1 into tNext
         end if
         if tNext > number of lines of gPlayerBaseList then put 1 into tNext
         put line tNext of gPlayerBaseList into  gGameObjectIndexLabel
         put char 3 to -1 of  gGameObjectIndexLabel into gID
         Base.select (gGameObjectIndexLabel)
         --Base.refreshGui gGameObjectIndexLabel
         UI.zoomedMap
         break
      case 65308 -- 'WHEEL DOWN = ZOOM OUT" 
         ops.setDebug ("65308 KEY" && tKey )
         UI.clearOffsetXOffsetY
         subtract 0.05 from gZoomLevel
         if gZoomLevel <= 0.05 then put 0.05 into gZoomLevel
         if gZoomLevel <= 0 then put 0 into gMapScale
         UI.ZoomedMap
         break
      case 65309 -- "WHEEL UP = ZOOM IN"
         ops.setDebug ("65309 KEY" && tKey )
         UI.clearOffsetXOffsetY
         add 0.05 to gZoomLevel
         if gZoomLevel >= 1.25 then put 1.25 into gZoomLevel
         UI.ZoomedMap
         break
      case 101 -- "E KEY" 
         put the currentSelection of this stack into gSelectedObject
         if word 1 of gSelectedObject is "base" then
            GameObject.Data  word 1 of gSelectedObject, word 2 of gSelectedObject
            GameObject.setPropertyValuesFromData	
            if gParent = 1 then UI.ToggleSendTo
         end if
         break
      case 117 -- "U KEY"
         UI.ToggleRendezvousToolBool
         break
         --65362 and tKey <= 65364 then pass RawKeyDown
      case 65361 --"LEFT ARROW"
         add -8 to gScrollOffsetX
         UI.zoomedmap
         break
      case 65363 --"RIGHT ARROW"
         add 8 to gScrollOffsetX
         UI.zoomedmap 
         break
      case 65362 -- "UP ARROW"
         add -8 to gScrollOffsetY
         Ui.zoomedmap 
         break
      case 65364 -- "DOWN ARROW"
         add 8 to gScrollOffsetY
         Ui.zoomedmap 
         break
   end switch
   focus button "theRawKey" of stack (the mainStack of this stack)
   pass rawKeyDown
end rawKeyDown

on Turn.Mobile
   ops.setDebug ("00187 Turn.Mobile")
   put "    " after gErrorIndent
   put Mobile.list() into gMobileObjectstList
   
   --- no defense satellites 
   put gMobileObjectstList into gFlightList 
   filter gFlistList without "d_"&"*"
   
   put getMobilePathsData() into gMobilePathData
   repeat for each line gMobileIndexLabel  in gMobileObjectstList
      if char 1 of gMobileIndexLabel = "b" then
         ops.setDebug ("ALERT A BASE" && gMobileIndexLabel &&"WAS FOUND IN  gMobileObjectstList via  Mobile.list()!!!")
         next repeat
      end if
      put " " after gErrorIndent   
      
      put Mobile.getData(gMobileIndexLabel) into gMobileData
      ops.setDebug ("00187 Turn.Mobile gMobileData =" && gMobileData)
      
      put "p" & char 2 to -1 of gMobileIndexLabel into gMobilePathIndexLabel
      ops.setDebug ("00187 Turn.Mobile   gMobilePathIndexLabel = " && quoted(gMobilePathIndexLabel))   
      
      put the travelIndex of graphic gMobileIndexLabel  into tTravelIndex
      ops.setDebug ("00187 Turn.Mobile   tTravelIndex = " && tTravelIndex)   
      
      
      ---  we need the ORIGIN POINTS of the path for ORIGINXY of the mobile
      set the customPropertySet of this stack to "PathData"
      do "put the" && gMobilePathIndexLabel &&"of this stack into gMobilePathPoints"
      ops.emptyDataSet
      ops.setDebug ("00187 Turn.Mobile   gMobilePathPoints = " && number of lines of gMobilePathPoints && "lines") 
      if number of lines of gMobilePathPoints < 2 then next repeat
      
      put the loc of graphic gMobileIndexLabel  into gMobileObjectLoc    
      ops.setDebug ("00187 Turn.Mobile" && quoted(gMobileIndexLabel) && "current Loc = " && gMobileObjectLoc)
      
      --- Mobile has reached end of its path
      if  tTravelIndex+1 > number of lines of gMobilePathPoints then
         ops.setDebug("Turn.Mobile" && quoted(gMobileIndexLabel) && "Destination reached!")
         put 0 into tTravelIndex
         set the travelIndex of graphic gMobileIndexLabel  to tTravelIndex
         
         put gMobileIndexLabel && "Arrived at Destination" && item sendtoitem of gMobileData into gMobileMessage
         set the customPropertySet of this stack to EventData
         do "set the" && gMobileIndexLabel && "of this stack to gMobileMessage" 
         ops.emptydataset
         ---GameObject.reversePath ---?
         --clearMobilePathIndex
         
         put "x" & the originXY of graphic gMobileIndexLabel into gCurrentMapLocation
         replace comma with "y" in gCurrentMapLocation
         replace "." with "p" in gCurrentMapLocation
         replace "-" with "neg" in gCurrentMapLocation
         set the customPropertySet of this stack to MapData
         do "put the" && gCurrentMapLocation && "of this stack into gCurrentMapLocationArray"
         put gMobileData into gCurrentMapLocationArray[gMobileIndexLabel]
         do "set the" && gCurrentMapLocation && "of this stack to gCurrentMapLocationArray"
         set the customPropertySet of this stack to empty
         
         next repeat
      end if
      --- LERP  along the line 
      if tTravelIndex <= number of lines of gMobilePathPoints then
         add 1 to tTravelIndex
         set the travelIndex of graphic gMobileIndexLabel  to tTravelIndex
         
         --- get the location of the mobile  coordinates on the current line of the Path points	
         --- get the originXY points of the line   and pass it to the mobile's originXY for zooming    
         put line tTravelIndex of gMobilePathPoints into gOriginXYLoc
         set the originXY of graphic gMobileIndexLabel to gOriginXYloc
         ops.setDebug ("00187 Turn.Mobile line" && tTravelIndex &&"of path" && gMobilePathIndexLabel &&"= gOriginXYLoc = " && gOriginXYLoc) 	
         
         --- set the mobile object to the current scaled points of the line
         put the points of graphic gMobilePathIndexLabel into gMobilePathScaledPoints
         set the loc of graphic gMobileIndexLabel  to line tTravelIndex of gMobilePathScaledPoints
         ops.setDebug ("MOBILE" && gMobileIndexLabel &&"MOVED TO = " && the loc of graphic gMobileIndexLabel)	         
         
         --put math.lerp(item 1 of tOriginLoc,item 1 of tDestination,tLerpProgress) into tNewX
         --put math.lerp(item 2 of tOriginLoc,item 2 of tDestination,tLerpProgress) into tNewY
         
         --- change the path to reflect the distance traveled, moving the first line forward
         --put number of lines of gMobilePathPoints into tDebugpathPoints   
         --ops.Setdebug ("Turn.Mobile changed gMobilePathPoints with tTravelIndex" && tTravelIndex && "number of lines of path points =" && tDebugpathPoints) 
         --put number of lines of gMobilePathPoints into tDebugpathPoints
         --ops.Setdebug ("Turn.Mobile changed gMobilePathPoints Number of lines =" && tDebugpathPoints &&"to"&& number of lines of gMobilePathPoints) 
      end if
      ops.emptyDataSet		
   end repeat
   -- UI.zoomedMap
end Turn.Mobile

--- only works for a straight line
function Math.subdividePolygon tStart, tEnd, tsubdivisions
   put item 1 of tStart into x1
   put item 2 of tStart into y1
   put item 1 of tEnd into x2
   put item 2 of tEnd into y2
   put x1 & "," & y1 & cr into myPolygonPoints
   repeat with i = 1 to tsubdivisions
      put (x1 + i * (x2 - x1) / tsubdivisions) & "," & (y1 + i * (y2 - y1) / tsubdivisions) & cr after myPolygonPoints
   end repeat
   filter lines of myPolygonPoints without empty
   return myPolygonPoints
end  Math.subdividePolygon

function Base.IncrementGoals tVal, tGoal, tmin, tmax
ops.setDebug ("00238 Base.IncrementGoals")
put "" into gErrorIndent
	   put "Base.IncrementGoals("&tGoal, tVal, tmin, tmax&") = " && tVal&cr after gDebug
	   if tGoal >= tVal then
	      add 3 to tVal
	      if tVal > tGoal then put tGoal into tVal
	      if tVal > tmax then put tMax into tVal
	      return tVal
   end if
	   
	   if tGoal <= tVal then
	      subtract 3 from tVal
	      if tVal < tGoal then put tGoal into tVal
	      if tVal < tMin then put tMin into tVal
	      return tVal
   end if
end Base.IncrementGoals

function ops.checkScriptForDuplicateFunctions

   put tScript into tHands
   put tScript into tFuncs
   filter lines of tHands with "on "&"*"
   filter lines of tFuncs with "function "&"*"
   put tHands & cr & tFuncs into tFuncHandList
   repeat  with i =  number of lines of tFuncHandList  to 1 step -1
      put line -1 of tFuncHandList into tCheck
      put tFuncHandList into tFiltered
      filter lines of tFiltered with tCheck
      if number of lines of tFiltered > 1 then put word 1 to 2 of tCheck && number of lines of of tFiltered & cr after tDuplicatesList
      filter lines of tFuncHandList without tCheck
   end repeat
   return tDuplicatesList
end ops.checkScriptForDuplicateFunctions

   if gBuild > 2 and mSendTo is not empty and mSendTo is not 0 and there is a graphic mSendTo then

function setOffsetXY x,y,tMapX,tMapY
ops.setDebug ("00302 setOffsetXY")

	   -- Calculate the offset between the selected object's initial location and gMapX,gMapY
	   put (x - tMapX) into gOffsetX
	   put (y - tMapY) into gOffsetY

	   put the customKeys of this stack into gMapObjects
	   --list.rendezvous
	   put Mobile.list() & cr & gRendezvousList & cr before gMapObjects

	   -- --Adjust the location of all the other objects on the map
	   repeat for each line tGraphicName in gMapObjects
	      if there is a graphic tGraphicName of this stack then
	         put the loc of graphic tGraphicName of this stack into tLoc
	         --put the originXY of graphic tGraphicName of this stack into tLoc
	         put (item 1 of tLoc - gOffsetX) into tNewX
	         put (item 2 of tLoc - gOFfsetY) into tNewY
	         set the loc of graphic tGraphicName of this stack to item 1 of tLoc - gOffsetX,item 2 of tLoc - gOffsetY
	         put loc of graphic tGraphicName of this stack is within the rect of group "map" of this stack into tVis
	         set the visible of graphic tGraphicName of this stack to tVis
      end if
   end repeat
	   --offset path data
	   set the customPropertySet of this stack to "PathData"
	   put the customKeys of this stack into gPathList
	   filter lines of gPathList without empty
	   put empty into tNewPoints
	   repeat for each line tPath in gPathList
	      do "put the" && tPath&&"of this stack into tPointsList"
	      filter lines of tPointsList without empty
	      put empty into tNewPoints
	      if tPointsList is not empty then
	         set itemDel to comma
	         repeat for each line tPoints in tPointsList
	            put (((item 1 of tPoints - tMapX) * gZoomLevel) + tMapX)- gOffsetX into tNewX
	            put (((item 2 of tPoints - tMapY) * gZoomLevel) + tMapY)- gOffsetY into tNewY
	            put tNewX &comma & tNewY & cr after tNewPoints
         end repeat
	         filter lines of tNewPoints without empty
      end if
	      if there is a graphic tPath then set the points of graphic tPath to tNewPoints
   end repeat
	   ops.emptyDataSet

	   return gOffsetX,gOffsetY

	   --put "tron,sark,clu,flynn,yori" into tempPlayerNames

function Turn.AssemblyUpdate tGameObjectIndexLabel
   ops.setDebug ("00309 Turn.AssemblyUpdate")
   put "" into gErrorIndent
   if tGameObjectIndexLabel is not empty then put tGameObjectIndexLabel into tGameObjectIndexLabel
   if gGameObjectIndexLabel is empty then
      ops.setDebug ("ERROR Turn.AssemblyUpdate received EMPTY gGameObjectIndexLabel")
      return empty
   end if
   ops.setDebug ("00309 Turn.AssemblyUpdate("&gGameObjectIndexLabel& ")")
   --- update ships assembly state
   set itemDel to comma
   GameObject.select gGameObjectIndexLabel  --- << --- update all globals associated with the object
   put "base" into gGameObjectTypeLabel
   put "baseData" into gGameObjectTypeData
   put false into bUpdateAIGoals	
   
   --- Assembly Complete add a new mobile
   
   if gAssembly+(gMotivation*0.15)+(gLife*0.0000000000123456789) > 100 then
      ops.setdebug ("ASSEMBLY gAssembly+(gMotivation*0.0015) > 100")
      --- reset assembly
      put 1 into gAssembly
      GameObject.assembly gAssembly   -- this should handle everything as far as gGameData etc
      	      
      --- NEW MOBILE
      put GameObject.createMobile(gGameObjectIndexLabel) into gMobileData --- << --- pass the source base data to the mobile with changes
      ops.setDebug ("ASSEMBLY RETURNS FROM GameObject.createMobile(" & quoted(gGameObjectIndexLabel) &")"&cr& "gGameObjectData ="&& gGameObjectData & cr &"gMobileData ="&& gMobileData )
      
      --- change AI Management Decision
      put true into bUpdateAIGoals	
      ops.emptyDataSet   
   else
      ops.setDebug ("!! ASSEMBLY BEFORE!" & gGameObjectIndexLabel && gAssembly) 
      add 1+(gProduction*0.15)+(gLife*0.0000000000123456789) to gAssembly
      ops.setDebug ("!! ASSEMBLY AFTER!" & gGameObjectIndexLabel && gASsembly) 
      GameObject.Assembly gAssembly      	      
      
      ops.setDebug ("Turn.AssemblyUpdate set gGameObjectData:" && gGameObjectData)
      set itemDel to comma
   end if
 
   ---  AI activity   after Aseembly update	   
   if bUpdateAIGoals is true and gID is not empty and gParent <>0 then Base.adjustAIGoalsAfterAssembly
   
   ops.setDebug ("Turn.AssemblyUpdate (" && gGameObjectIndexLabel &&") returns gGameObjectData" && gGameObjectData & cr)
   return gGameObjectData
end Turn.AssemblyUpdate
