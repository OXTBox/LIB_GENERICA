on extractUI

   if there is not a folder tStackFolder then create folder tStackFolder

      put the long ID of control i of this card into tID
      put word 1 of tID into tObjType
      put tStackFolder &"/"  &tObjType into tObjFolder
      if there is not a folder tObjFolder then create folder tObjFolder
      put tObjFolder  & "/" &tObjType&"_Scripts" into tSFolder

      end if 
      put tID &cr into tPropOut
      put the short name of control i of this card into tShorty
      put the properties of tID into tPropertiesA
      repeat for each line tProp in the keys of tPropertiesA
         put tPropertiesA[tProp] into tPropVal
         put tProp &"=" & tPropVal & cr after tPropOut
      end repeat
      put tPropOut into URL ("file:" & tObjFolder & "/" & tShorty & ".txt")
   end repeat
   
   put tStackFolder & "/Stack_Properties" into tPFolder
   if there is not a folder tPFolder then create folder tPFolder

      set the customPropertySet of this stack to PSet
      put the customKeys of this stack into tKeys
      repeat for each line K in tKeys
         put tPsetFolder &"/" & K into tKeyFolder

            if the K of this stack is not empty or (the number of lines of the keys of (the k of this stack) > 0) then

                     put tKeyFolder & "/" & KK into subKeyFolder
                     if there is not a folder subKeyFolder then create folder subKeyFolder
                     put tArrayA[KK] into URL ("file:" & subKeyFolder &"/" & Kk &".txt")
                  end repeat
               else
                  put (the k of this stack)into URL ("file:" & tKeyFolder &"/" & K &".txt")
               end if
               --put jsonExport(the K of this stack) into URL ("file:" & tKeyFolder &"/" & K &".txt")
            end if
         end try
      end repeat
   end repeat
   
   set the customPropertySet of this stack to empty
   put the customKeys of this stack into tKeys

         if the K of this stack is not empty  then put jsonExport(the K of this stack) into URL ("file:" & tKeyFolder &"/" & K &".txt")
         
      end try
