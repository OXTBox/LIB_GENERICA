on ops.addErrorCodesToStackScript
   ops.setDebug ("00148 ops.addErrorCodesToStackScript")
   put "" into gErrorIndent
   set the cursor to busy
   put the script of this stack into tScript
   put tScript into tHands
   put tScript into tFuns
   filter lines of tHands with "on "&"*"
   sort tHands
   put 0 into tHandlerCount
   put "" into tModifiedStackScript
   repeat for each line tHand in tHands
      add 1 to tHandlerCount
      put "" into tHandlerBody
      put word 2 of tHand into tHandler
      	      
      put getHandlerFromStack(tHandler) into tHandlerBody
      filter lines of tHandlerBody without empty
      if the number of words of tHand > 2 then
         put "" into tArgs
         repeat for each item I in (word 3 to -1 of tHand)
            if char 1 of I is "-" then put ")" &I && "&&" after tArgs
            else
               put  I && "&&" after tArgs
            end if
         end repeat 
         put space & "&&" && char 1 to -3 of tArgs  after tErrorCodeLine
      end if
      put   line 1 of tHandlerBody & cr & tErrorCodeLine &")"  & cr into tHandlerUpate
      put  "put" && quote & quote && "into gErrorIndent" & cr & line 2 to -1 of tHandlerBody after tHandlerUpate
      if word 1 of line 1 of tHandlerUpate is numtochar(111)&numtochar(110)   then put tHandlerUpate & cr & cr after tModifiedStackScript
      if word 1 of line 1 of tHandlerUpate is "function"  then put tHandlerUpate & cr & cr after tModifiedStackScript
      string.writeFile "Handlers",word 2 of tHand,tHandlerUpate
   end repeat
   string.writeFile "StackScript","Handlers",tModifiedStackScript
   -- functions
   filter lines of tFuns with "function "&"*"
   sort tfuns
   repeat for each line tFunc in tFuns
      add 1 to tHandlerCount
      put "" into tHandlerBody
      put word 2 of tFunc into tFunctionLabel
      put getFunctionFromStack(tFunctionLabel) into tHandlerBody
      if the number of words of tHand > 2 then
         put "" into tArgs
         repeat for each item I in (word 3 to -1 of tHand)
            put  I && "&&" after tArgs
         end repeat 
         put space & "&&" && char 1 to -3 of tArgs  after tErrorCodeLine
      end if
      put  tErrorCodeLine &")" after char -1 of line 1 of tHandlerBody
      put cr &"put" && quote &quote & "into gErrorIndent" after line 2 of tHandlerBody
      replace space&space with empty in line 1 of   tHandlerBody
      put tHandlerBody & cr & cr after tModifiedStackScript
      string.writeFile "Functions",tFunctionLabel,tHandlerBody
   end repeat
   	   
   put tScript into tGlobals
   filter lines of tGlobals with "global"&"*"
   sort tGlobals
   replace tab with space in  tglobals
   replace space & space & space with space in tGlobals
   replace space & space & space with space in tGlobals
   replace space & space with space in tGlobals
   sort tGlobals
   put tGlobals & cr & "----END OF GLOBAL DEFINITIONS----"  into tGlobalsText
   put tGlobalsText & cr & cr before tModifiedStackScript
   string.writeFile "StackScript","StackScript_ErrorCodesAdded_"&string.asitems(the short date,"/","_"),tModifiedStackScript
   	   
   repeat for each line G in tGlobals
      put "put" && quote&quote &&"into" && G & cr after tBigInit
   end repeat
   	   
   put "on INIT.All_GLOBALS" & cr before tBigInit
   put  "end INIT.ALL_GLOBALS" after tBigInit
   put tGlobalsText & cr & cr before tBigInit
   string.writeFile "Globals","GlobalInit__"&string.asitems(the short date,"/","_"),tBigInit
   set the cursor to arrow
end ops.addErrorCodesToStackScript

function StackScriptToCustomProperties
ops.setDebug ("00303 StackScriptToCustomProperties")
put "" into gErrorIndent
	   put the script of this stack into tStackscript
	   put tStackscript into tStackHandlers
	   put tStackscript into tStackFunctions
	   filter lines of tStackHandlers with ("on"&numtochar(32)&"*")
	   filter lines of tStackFunctions with ("function"&numtochar(32)&"*")
	   put tStackHandlers & cr & tStackFunctions into tAllStackHandHeaders
	   --put tAllStackHandHeaders
	   sort tAllStackHandHeaders
	   filter lines of tAllStackHandHeaders without empty
	   filter lines of tAllStackHandHeaders without tab&"*"
	   filter lines of tAllStackHandHeaders without "*"&"put"&"*"
	   
	   repeat for each line tGetBeginLineString in tAllStackHandHeaders
	      put lineoffset(tGetBeginLineString,tStackscript)-1 into tFunctionHeaderOffset
	      if tFunctionHeaderOffset is 0 then next repeat
	      put line tFunctionHeaderOffset+1 of tStackscript into tActualFunctionLabeLine
	      put lineoffset("end" && word 2 of tActualFunctionLabeLine,tStackscript)+1 into tFunctionFooterOffset
	      if tFunctionHeaderOffset is 0 then next repeat
	      put line tFunctionHeaderOffset to tFunctionFooterOffset of tStackscript into tCurrentFunction
	      if CurrentFunction is empty then next repeat	
	      put line tFunctionHeaderOffset+1 of tStackscript into tActualFunctionLabeLine
	      put tActualFunctionLabeLine & cr after allFunctionsHeaderLabels
	      put word 2 of tActualFunctionLabeLine into tFunctionName
	      replace "on " with empty in tFunctionName
	      replace quote with empty in tFunctionName
	      if word 1 of tActualFunctionLabeLine is "function" then
	         set the customPropertySet of this stack to StackFunctions
	         put the currentFunctions of this stack into tArray
	         put tCurrentFunction into tArray[tFunctionName]
	         SET the currentFunctions of this stack to tArray	
	         ops.emptyDataSet
	      else
	         			
	         set the customPropertySet of this stack to StackHandlers
	         put the currentHandlers of this stack into tArray
	         put tCurrentFunction into tArray[tFunctionName]
	         SET the currentHandlers of this stack to tArray	
	         ops.emptyDataSet
      end if	
   end repeat
	   replace comma with tab & comma in allFunctionsHeaderLabels
	   sort allFunctionsHeaderLabels
	   return (allFunctionsHeaderLabels)
end StackScriptToCustomProperties

function filterStackScriptForGlobals tGlobalPrefix
   if tGlobalPrefix is an integer then put numtochar(tGlobalPrefix) into tGlobalPrefix
   if tGlobalPrefix is  empty then put "g" into tGlobalPrefix
   put tGlobalPrefix into gGlobalPrefixChar
   put the script of this stack into tScript
   put tScript into tGlobals
   filter tGlobals with "global " &"*"
   --filter tscript without "global " &"*"
   repeat with i = 1 to 64
      if  i < 10 or i > 11  or i <> 46 or i <> 95 then
         replace numtochar(i) with cr in tScript
      end if
   end repeat
   filter lines of tScript without empty
   repeat with i = 91 to 255
      if i < 97 or i > 122 and i <> 46 and i <> 95 then
         replace  numtochar(i) with cr in tScript
      end if
   end repeat
   filter lines of tScript with "*"&gGlobalPrefixChar&"*"
   filter lines of tScript with gGlobalPrefixChar&"*"
   put "" into gFoundGlobals
   repeat with i = number of lines of tScript down to 1
      put line i of tscript into tGlobalName
      filter  lines of tScript  without tGlobalName
      -- only matching character for globals gGlob DO_SOMETHING etc
      if chartonum(char 1 of tGlobalName) = chartonum(char 1 of gGlobalPrefixChar) then
         if chartonum(char 2 of tGlobalName) >= 65 and chartonum(char 2 of tGlobalName) <=  90 then 
            put tGlobalName & cr after gFoundGlobals
         end if
      end if
   end repeat
   filter lines of gFoundGlobals without empty
   sort gFoundGlobals
   return gFoundGlobals
end filterStackScriptForGlobals

on objectPropertyDimensionChanged tFocusedField
   put the customPropertySet of this card into gCurrentPropertySet 
   set the customPropertyset of this card to Game_UI_Properties
   put "width,height,x,y,left,top,right,bottom" into gButtonPropertyGeometryFields
   set itemDel to comma
   lock screen
   lock messages
   repeat for each item I in gButtonPropertyGeometryFields
      do "put field"&& quote & I&   "field" &quote&&"into g" &I
   end repeat
   switch tFocusedField
      case "widthField"
         put gLeft+gWidth into gRight
         put gRight into field "rightField"
         put gWidth/2 into gX
         put gX into field "xField"
         break
      case "heightField"
         put gTop+gHeight into gBottom
         put gBottom into field "bottomField"
         put gHeight/2 into gY
         put gY into field "xField"
         break
         
      case "leftField"
      case "rightField"
         put  gRight - gLeft  into gWidth
         put gWidth into field "wdithfield"
         put gLeft+gWidth /2 into gX
         put gX into field "xField"
         break
         
      case "topField"
      case "bottomField"
         put gTop - gBottom into gHeight
         put gHeight into field "heightfield"
         put gTop+gHeight /2 into gY
         put gY into field "yField"
         break
   end switch
   unlock messages
   unlock screen
   updateUIPropertiesDataFromControls
   set the customPropertySet of this card to gCurrentPropertySet
end objectPropertyDimensionChanged
