on setCardScript
   ops.setDebug ("00165 setCardScript")
   put "" into gErrorIndent
   put the script of this stack into tStackScript
   put tStackScript into tFilteredList
   filter lines of tFilteredList with "on SCRIPT_CARD_"&"*"
   --put tFilteredList
   repeat for each line L in tFilteredList
      put lineOffset(word 2 of L,tStackScript) into tStart
      put lineOffset("end" && word 2 of L,tStackScript) into tEnd
      put line tStart to tEnd of tStackScript & cr & cr after tCardScript
   end repeat
   replace "SCRIPT_CARD_" with empty in tCardScript
   --put tCardScript
   lock messages
   set the script of this card to tCardScript
   unlock messages
end setCardScript

   put lineoffset("on ",tHandler) into n

   if n = 0 then exit setHandlerEditHeader
   put word 2 of line n of tHandler into tHandlerName
   	   
   set the customPropertySet of this stack to StackEdits
   	   
   put the number of lines of the customKeys of this stack + 1 into gHandlerEditCount
   	   
   put "-"&"-"&"-"&"-" into tEditingLinePrefix
   filter lines of tHandler without tEditingLinePrefix&"*"
   put toupper(tHandlerName) into tHandlerCaps

   -- ONE LINE for the ending
   if number of lines of tHandler = 1 then
      put tEditingLinePrefix &"END"&tHandlerCaps&tEndOfEditFooter into gGameClipboardText
      set the clipBoardData["text"] to gGameClipboardText
   else
      	      
      put the customKeys of this stack into gEditHistoryList

      put number of lines of gEditHistoryList+1 into gHandlerVistedCount
      put tHandlerName&"_"&string.PadNumber(gHandlerVistedCount,3) into gCurrentEditIndexLabel

      if word 1 of line n+1 of tHandler is not "ops.setDebug" then
         put line 1 to n of tHandler& cr &"ops.setDebug" &&quote& tHandlerName &quote& cr & line n+1 to -1 of tHandler into tHandler

      -- OR WHOLE HANDLER
      filter lines of tHandler without "--" & "--" & "*"
      put tEditingLinePrefix & tHandlerCaps && "_EDITING @" & gCurrentEditIndexLabel into tEditHeader
      put tEditHeader & cr & tHandler & cr & tEndOfEditFooter into gGameClipboardText
      	      
   end if
   put quote & tHandlerName &tab& the long time & the long date & quote into tEditNote
   --put tEditNote into tEditPropertyArray[gCurrentEditIndex]
   --set the Edits of this stack to tEditPropertyArray
   do "set the" && gCurrentEditIndexLabel &&"of this stack to tEditNote"

      put the loc of graphic  gGameObjectIndexLabel of stack (the mainStack of this stack) into tLoc

      add item 2 of tLoc to gMapCenterY

   put gMapCenterX & cr & gMapCenterY & cr after gDebug

function setOffsetXY x,y,tMapX,tMapY
ops.setDebug ("00302 setOffsetXY")
put "" into gErrorIndent
	   -- Calculate the offset between the selected object's initial location and gMapX,gMapY
	   put (x - tMapX) into gOffsetX
	   put (y - tMapY) into gOffsetY
	   set the customPropertySet of this stack to "BaseData"
	   put the customKeys of this stack into gMapObjects
	   --list.rendezvous
	   put Mobile.list() & cr & gRendezvousList & cr before gMapObjects
	   filter lines of gMapObjects without empty
	   -- --Adjust the location of all the other objects on the map
	   repeat for each line tGraphicName in gMapObjects
	      if there is a graphic tGraphicName of this stack then
	         put the loc of graphic tGraphicName of this stack into tLoc
	         --put the originXY of graphic tGraphicName of this stack into tLoc
	         put (item 1 of tLoc - gOffsetX) into tNewX
	         put (item 2 of tLoc - gOFfsetY) into tNewY
	         set the loc of graphic tGraphicName of this stack to item 1 of tLoc - gOffsetX,item 2 of tLoc - gOffsetY
	         put loc of graphic tGraphicName of this stack is within the rect of group "map" of this stack into tVis
	         set the visible of graphic tGraphicName of this stack to tVis
      end if
   end repeat
	   --offset path data
	   set the customPropertySet of this stack to "PathData"
	   put the customKeys of this stack into gPathList
	   filter lines of gPathList without empty
	   put empty into tNewPoints
	   repeat for each line tPath in gPathList
	      do "put the" && tPath&&"of this stack into tPointsList"
	      filter lines of tPointsList without empty
	      put empty into tNewPoints
	      if tPointsList is not empty then
	         set itemDel to comma
	         repeat for each line tPoints in tPointsList
	            put (((item 1 of tPoints - tMapX) * gZoomLevel) + tMapX)- gOffsetX into tNewX
	            put (((item 2 of tPoints - tMapY) * gZoomLevel) + tMapY)- gOffsetY into tNewY
	            put tNewX &comma & tNewY & cr after tNewPoints
         end repeat
	         filter lines of tNewPoints without empty
      end if
	      if there is a graphic tPath then set the points of graphic tPath to tNewPoints
   end repeat
	   ops.emptyDataSet
	   -- Return the offset values
	   return gOffsetX,gOffsetY
end setOffsetXY
