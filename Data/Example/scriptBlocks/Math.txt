function Math.subdividePolygon tStart, tEnd, tsubdivisions
   put item 1 of tStart into x1
   put item 2 of tStart into y1
   put item 1 of tEnd into x2
   put item 2 of tEnd into y2
   put x1 & "," & y1 & cr into myPolygonPoints
   repeat with i = 1 to tsubdivisions
      put (x1 + i * (x2 - x1) / tsubdivisions) & "," & (y1 + i * (y2 - y1) / tsubdivisions) & cr after myPolygonPoints
   end repeat
   filter lines of myPolygonPoints without empty
   return myPolygonPoints
end  Math.subdividePolygon

function Math.DeltaToTarget x1,y1,x2,y2
ops.setDebug ("00280 Math.DeltaToTarget")
put "" into gErrorIndent
	   repeat with i = 0 to 1 step 0.01
	      put x1 +(x2 - x1) *i into newX
	      put y1 +(y2 - y1) *i into newY
	      put newX & comma & newY & cr after tPathToTarget
   end repeat
	   --// use the first and last line of the point list to determine the x,y delta
	   put line 1 of tPathToTarget into axy
	   put line -1 of tPathToTarget into bxy
	   put (item 1 of bxy - item 1 of axy)/100 into dx
	   put (item 2 of bxy - item 2 of axy)/100 into dy
	   return dx,dy
end Math.DeltaToTarget
