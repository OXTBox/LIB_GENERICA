on Init.RandomBases
   ops.setDebug ("00114 Init.RandomBases")
   put "" into gErrorIndent
   set itemDel to comma
   put item 1 to -4 of Base.setzeroGlobals("globals") into gZeroedGameObjectData 
   put 0,0 into gBaseMapCenterXY
   put 0 into gBaseMapCenterX
   put 0 into gBaseMapCenterY
   put 0 into gNumberOfGameObjects
   set the customPropertySet of this stack to "BaseData"
   set the customKeys of this stack to empty
   --ops.emptyDataSet
   set the randomSeed to the seconds
   if gMapDensityStep < 32 then put 16 into gMapDensityStep
   repeat with gx = (gMapX-1024) to gMapX+1024 step gMapDensityStep
      repeat with gy = (gMapX-1024) to gMapX+1024 step gMapDensityStep
         	   
         put random(100) into rnd1
         put random(100) into rnd2
         put rnd1 & cr & rnd2 & cr after gRandomNumbersList
         if rnd1 >88 and rnd2 > 88 then 
            add 1 to gNumberOfGameObjects
            --- do this before setting globals in the handler GameObject.createBase!!
            put gZeroedGameObjectData into gGameObjectData -- fall back to zeroed index to have a place to put things	
            put string.PadNumber(gNumberOfGameObjects,3) into gID
            -- CREATE GRAPHIC OBJECT HERE
            GameObject.createBase gID,gx,gy,"base"
            add gx to gBaseMapCenterX
            add gy to gBaseMapCenterY
         end if
      end repeat
   end repeat
   put getBaseMapCenterXY() into gBaseMapCenterXY
   put "Init.RandomBases done @" && the long time& cr after gDebug
end Init.RandomBases
