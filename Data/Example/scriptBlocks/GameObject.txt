function GameObject.createMobile gGameObjectIndexLabel 
   ops.setDebug ("00246 GameObject.createMobile")
   put "" into gErrorIndent
   ops.setdebug ("$$ GameObject.createMobile received"&&quoted(gGameObjectIndexLabel) )
   
   --- GET THE SOURCE BASE DATA THAT MADE THE MOBILE
   --- 02-09-2024 ASSUME WE HAVE THE DATA
   --- GameObject.Select gGameObjectIndexLabel    
   	 
   --- CREATE THE MOBILE ID  
   add 1 to gNumberOfGameObjects
   put string.PadNumber(gNumberOfGameObjects,3) into gMobileID
   	   
   --- GET THE BUILD TYPE AND PREFIX
   ops.setdebug("GameObject.createMobile gBuild =" && gBuild)
   if gBuild < 2 or gBuild is empty then put 2 into gBuild    -- gBuild should not be 1 or 'base'!!
   put item gBuild of gGameObjectSelectables into gMobileTypeIndexLabel
   
   put char 1 of gMobileTypeIndexLabel  into gMobileTypePrefix
   ops.setdebug("GameObject.createMobile gBuild =" && gBuild)
   ops.setdebug("GameObject.createMobile gMobileTypePrefix =" && gMobileTypePrefix)	   
   
   --- CREATE THE NAME OF THE MOBILE
   put gMobileTypePrefix &"_"& gMobileID  into gMobileIndexLabel
   ops.setdebug("GameObject.createMobile gMobileIndexLabel =" && gMobileIndexLabel)	 
   
   --- PASS THE BASE DATA TO THE MOBILE DATA
   put gGameObjectData into gMobileData
   
   --- SET THE ID OF THE MOBILE OBJECT TO  NUMBER OF OBJECTS
   put gMobileID into item 1 of gMobileData
   
   --- CHANGE THE OBJECT TYPE TO THE BUILD TYPE
   put gBuild into item typeitem of gMobileData
   
   ops.setdebug("GameObject.createMobile gMobileID =" && gMobileID)	 
   	   
   ---  ADD MOBILE OBJECT TO PROWLERDATA ARRAY  -- maybe not
   --- Base.AssignProwler
   
   ops.setDebug("GameObject.createMobile setting MobileData of gMobileIndexLabel" && gMobileIndexLabel)
   
   -- ADD MOBILE OBJECT TO MOBILEDATA ARRAY
   if char 1 of gMobileIndexLabel is "d" then 
      set the customPropertySet of this stack to  ProwlerData
      do "put the" && gGameObjectIndexLabel  && "of stack (the mainStack of this stack) into gMobileDataArray"
      put gMobileData into gMobileDataArray[gMobileIndexLabel]
      do "set the" && gGameObjectIndexLabel  && "of stack (the mainStack of this stack) to gMobileDataArray"
   else
      set the customPropertySet of this stack to  MobileData
      do "set the" && gMobileIndexLabel  && "of stack (the mainStack of this stack) to gMobileData"
   end if
   
   
   
   
   ops.emptyDataSet
   
   ops.setDebug("GameObject.createMobile setting PlayerData to gMobileIndexLabel" && gMobileIndexLabel)
   -- STORE THE DATA TO THE PLAYER
   set the customPropertySet of this stack to "PlayerData"
   GameObject.getPropertyValuesFromData
   put "Player"& gParent into gPlayerName
   do "put the" && gPlayerName && "of this stack into gPlayerDataArray"
   put gMobileData into gPlayerDataArray[gMobileIndexLabel] 
   do "set the" && gPlayerName && "of this stack to gPlayerDataArray"
   ops.emptyDataSet
   
   --- IF MOBILE OBJECT IS NOT "DEFENSE" THEN CREATE A GRAPHIC FOR IT	   
   if gBuild > 2 then UI.createMobileGraphic gGameObjectIndexLabel
   
   put item SendToItem of gMobileData into mSendTo
   
   --- IF MOBILE OBJECT IS NOT "DEFENSE" THEN CREATE A PATH GRAPHIC FOR IT	 
   if gBuild > 2 and mSendTo is not empty and mSendTo is not 0 and there is a graphic mSendTo then
      -- CREATE A PATH AT THE TIME OF MOBILE CREATION TO PROCEED ALONG THE SENDTO PATH
      UI.createPathFromSendTo 
   end if
   ops.setDebug("__GameObject.createMobile complete with " &&  gMobileIndexLabel  &&"&"&& gMobilePathIndexLabel)
   return  gMobileData
end GameObject.createMobile

function GameObject.getIndexInvalid targvalue,targLabeL,tSource

	   put "" into gGameObjectIndexInvalid

	   put "" into gErrorType

	   if (targvalue is a number) then put "IS A NUMBER" into gErrorType
	   if (targvalue = 0) then put "IS A ZERO " into gErrorType
	   if (char -1 of targvalue is not a number) then put "NO NUMERAL ENDING" into gErrorType
	   if (char 1 of targvalue is a number) then put "STARTS WITH NUMBER" into gErrorType

      end repeat
   end if
	   
	   if (targvalue is empty) then put "IS A EMPTY" into gErrorType
	   if gErrorType is not empty then
	      put "-" &"-" &"-! ERROR @" && tSource && "variable" && targLabeL &tab& gErrorType into gErrorText
	      ops.setDebug (gErrorText)
	      put true into bGameObjectIndexInvalid
   end if
	   return bGameObjectIndexInvalid
end GameObject.getIndexInvalid

function GameObject.PropertyDetails tArg,tVal
ops.setDebug ("00256 GameObject.PropertyDetails")
put "" into gErrorIndent
	   switch tArg
	      case empty
	      case default
	         return gGameObjectProperties
	         break
	      case "get"
	         if tval is not empty then 
	            put itemOffset(tVal,gGameObjectProperties) into tOffset	
	            return (item tOffset of gGameObjectData)
	         else 
	            return gGameObjectData
         end if
	         break
	      case "offset"
	         return itemOffset(tVal,gGameObjectProperties)
	         break
	      case "name"
	         return item tVal of gGameObjectProperties
	         break
	      case "list"
	         put gGameObjectProperties into tList
	         replace comma with cr in tList
	         return tList
	         break
	      case "globals"
	         if tVal is not empty then put GameObject.setTypeLabel(tVal) into gGameObjectTypeLabel 
	         put "g"&gGameObjectTypeLabel & gGameObjectProperties into tList
	         replace comma with cr&"g"&gGameObjectTypeLabel in tList
	         return tList
	      case "current"
	         repeat for each item i in gGameObjectProperties
	            put "put" && i & "&&" &  "g"&i & "&&cr after tResult" & cr after GET_GAME_OBJECT_CURRENT_GLOBAL_VALUES
         end repeat
	         do GET_GAME_OBJECT_CURRENT_GLOBAL_VALUES
	         return tResult
	         break
	      case "display"
	         repeat for each item i in gGameObjectProperties
	            put "put g"&i & "&&cr after tResult" & cr after DISPLAY_GAME_OBJECT_CURRENT_GLOBAL_VALUES
         end repeat
	         
	         do DISPLAY_GAME_OBJECT_CURRENT_GLOBAL_VALUES
	         return tResult & cr & DISPLAY_GAME_OBJECT_CURRENT_GLOBAL_VALUES
	         
	         break
   end switch
	   return empty
end GameObject.PropertyDetails
