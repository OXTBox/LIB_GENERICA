on UI.ZoomedMap
ops.setDebug ("00225 UI.ZoomedMap")
put "" into gErrorIndent
	   lock screen
	   set the customPropertySet of this stack to "BaseData"
	   put the customKeys of this stack into gBaseList
	   ops.emptyDataSet
	   put the currentSelection of this stack into gSelectedObject
	   -- change center object later for zooming around to each selection
	   if there is not a graphic "centerxy" then UI.createCenterXYgraphic
	   put "centerxy" into gCenterObject
	   
	   if gSelectedObject is not empty  then  put word 2 of  gSelectedObject into gCenterObject
	   
	   -- get the center of the map
	   put gMapX into tMapX
	   put gMapY into tMapY
	   -- get the scroll offset of the map
	   add gScrollOffsetX to tMapX
	   add gScrollOffsetY to tMapY
	   	 	   
	   GameObject.MobileList
	   list.rendezvous
	   
	   put Mobile.list() & cr & gBaseList & cr &gRendezvousList into tZoomList
	  -- put gBaseList into tZoomList
	   filter lines of tZoomList without empty
	   repeat for each line tGraphicName in tZoomList
	      set itemDel to comma
	      -- Calculate the new position based on the zoom level
	      if there is a graphic tGraphicName of this stack then 
	         put the originXY of graphic tGraphicName of this stack into tLoc
	         put ((item 1 of tLoc - tMapX) * gZoomLevel) + tMapX into tNewLocX
	         put ((item 2 of tLoc - tMapY) * gZoomLevel) + tMapY into tNewLocY
	         -- Set the new location
	         set the loc of graphic tGraphicName of this stack to tNewLocX,tNewLocY
	         -- set visibility within the map group
	         put loc of graphic tGraphicName of this stack is within the rect of group "map" of this stack into tVis
	         set the visible of graphic tGraphicName of this stack to tVis
	         if tGraphicName is not word 2 of gSelectedObject and lineSize of graphic tGraphicName >1 then set the lineSize of graphic tGraphicName to 1
      end if
   end repeat
	   
	   
	   if gzoomLevel <0.26 or gCenterObject is empty then put loc of group "Map" into tRootLoc
	   put the loc of group "Map" into tRootLoc
	   if  there is a graphic gCenterObject then  put the loc of graphic gCenterObject into tRootLoc
	   -- center everything on the selection
	
	   put setOffsetXY(item 1 of tRootloc,item 2 of tRootLoc,tMapX,tMapY) into gOffsetXY
	   
	   
	   put word 1 of gSelectedObject into gGameObjectTypeLabel
	   put char 1 of gGameObjectTypeLabel into gGameObjectTypePrefix
	   put char 3 to -1 of word 2 of gSelectedObject into gID
	   put word 2 of gSelectedObject into gGameObjectIndexLabel
	   --
	   
	   
	   if  gGameObjectIndexLabel is not empty then
	      -- for some reason the rendezvous need to be adjusted outside of the original base map
	      -- else the rendezvous gets thrown into a strange location
	      -- which only occurs if setOffsetXY() is used to center the map on the selected object
	      fixWanderingRendezvous
	      UI.refreshSendTo
	      set the visible of graphic "SendToPath" to (gSendTo is not 0 and gSendTo is not empty and gParent = 1)
   end if
	   put gZoomLevel into field "zoomlevel"
	   GameObject.selectionCircle
	   unlock screen
end UI.ZoomedMap
