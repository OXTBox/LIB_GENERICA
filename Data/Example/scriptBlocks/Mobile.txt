on Mobile.refreshGUI
   ops.setDebug ("00137 Mobile.refreshGUI")
   put "" into gErrorIndent
   put "Mobile.refreshGUI" && gMobileIndexLabel  & cr after gDebug
   put gMobileIndexLabel  into field "SelectedMobileLabel"
   set the customPropertySet of this stack to MobileData
   do "put the" && gMobileIndexLabel && "of this stack into gMobileDataArray"
   ops.emptyDataSet
   put gMobileIndexLabel into gMobilePathIndexLabel
   put "p" into char 1 of gMobilePathIndexLabel
   set the customPropertySet of this stack to "PathData"
   do "put the" && gMobilePathIndexLabel && "of this stack into gMobilePathOriginPoints"
   ops.emptyDataSet
   put floorData(gMobilePathOriginPoints) into field "MobileListField"
   put "Life:" && item itemOffset("life",gGameObjectProperties) of gGameObjectData into field "MobileLifeLabel"
   put "Level:" && item itemOffset("level",gGameObjectProperties) of gGameObjectData into field "MobileLifeLabel"
   put gMobileDataArray[gMobileIndexLabel ] into field "GameObjectProperties"
   set the disabled of button "ClearPathPoint" to (selectedLine of field "MobileListField" is empty)
   replace comma with cr in field "GameObjectProperties"
end Mobile.refreshGUI

function Mobile.List 
   ops.setDebug ("00295 Mobile.List")
   put "" into gErrorIndent
   set the customPropertySet of this stack to MobileData
   put the customKeys of this stack into gMobileObjectList
   ops.emptyDataSet
   filter gMobileObjectList without "d_*"
   put gMobileObjectList into checkForBs
   filter checkForBs with "b_*"
   if  checkForBs is not empty then ops.setDebug("ALERT BASE FOUND IN gMobileObjectList via Mobile.List!!" &cr& checkForbs &cr)
   filter gMobileObjectList without "b_*"   -- no bases unless we want bases
   return gMobileObjectList
end Mobile.List
