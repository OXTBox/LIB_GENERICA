function string.directorySafeDate
ops.setDebug ("00305 string.directorySafeDate")
put "" into gErrorIndent
	   put the short date into gDate
	   set itemDel to "/"
	   put "" into gDirSafeTime
	   repeat for each item I in gDate
	      put string.padNumber(I,2) &"_" after gDirSafeTime
   end repeat
	   set itemDel to comma
	   return char 1 to -2 of gDirSafeTime
end string.directorySafeDate

function string.padNumber tNum,tLen
--- ops.setDebug ("00306 string.padNumber") !! NOISE
put "" into gErrorIndent
	   put number of characters of tNum into tCount
	   repeat with i = 1 to tLen-tCount
	      put 0 before tNum
   end repeat
	   return tNum
 end string.padNumber

function string.padString tString,tPadText,tLen
