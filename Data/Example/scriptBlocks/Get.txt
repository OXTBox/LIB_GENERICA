function getCommands tFilter
ops.setDebug ("00264 getCommands")
put "" into gErrorIndent
	   put the script of this stack into tScript
	   ops.setDebug "zzCollectScript" && (short name of the target)
	   
	   put tScript into gStackHandlers
	   put tScript into gStackFunctions
	   if tFilter is empty then
	      filter lines of gStackHandlers with ("on"&numtochar(32)&"*")
	      filter lines of gStackFunctions with (cr&"function"&numtochar(32)&"*")
	   else
	      filter lines of gStackHandlers with (cr&"on"&numtochar(32)&tFilter&"*")
	      filter lines of gStackFunctions with (cr&"function"&numtochar(32)&tFilter&"*")
   end if
	   replace cr with "()"& cr in gStackFunctions
	   put gStackHandlers & cr & gStackFunctions into gStackCommands
	   replace "function"& space with empty in gStackCommands
	   replace "on" & space with empty in gStackCommands
	   
	   sort gStackCommands
	   return gStackCommands
end getCommands

function getHandlerFromStack tHandlerName
ops.setDebug ("00267 getHandlerFromStack")
put "" into gErrorIndent
	   put the script of this stack into tScript
	   put lineOffset("on"&&tHandlerName,tScript) into tStart
	   put lineOffset("end"&& tHandlerName,tScript) into tEnd
	   put line tStart to tEnd of tScript into tHandlerChunk
	   repeat for each line L in tHandlerChunk
	      if word 1 of L is not "on" and word 1 of L is not "end" then
	         put tab & L & cr after tChunk
	      else
	         put L & cr after tChunk
      end if
	      put tChunk into tHandlerChunk
   end repeat
	   return tHandlerChunk
end getHandlerFromStack

function GetRelativeLoc pScreenLoc, pZoomLevel
   ops.setDebug ("00269 GetRelativeLoc  pScreenLoc, pZoomLevel" &&  quoted(pScreenLoc), pZoomLevel)
   put "" into gErrorIndent
   if pZoomLevel is empty or pZoomLevel is 0 then put gZoomLevel into pZoomLevel
   -- Get the center of the map
   --   if gCenterObject is not empty and gCenterObject is not "map" then 
   --      put the originXY of graphic gCenterObject into tMapCenter
   --   else
   --      put loc of group "map" into tMapCenter
   --   end if
   put loc of group "map" into tMapCenter
   put item 1 of tMapCenter into tMapX
   put item 2 of tMapCenter into tMapY
   -- Calculate the relative location
   put item 1 of pScreenLoc into tScreenX
   put item 2 of pScreenLoc into tScreenY
   --put tScreenX & cr & tScreenY
   put ((tScreenX-tMapX) / pZoomLevel )  into tRelativeX
   put ((tScreenY-tMapY) / pZoomLevel ) into tRelativeY
   ops.setDebug ("GetRelativeLoc( "&&pScreenLoc &&","&pZoomLevel&") ="&& tRelativeLoc )
   return tRelativeX & "," & tRelativeY
end GetRelativeLoc

on getAllCurrentGlobalValues
   filter tScript with "global "&"*"
   replace  "global" with empty in tScript
   put "" into gGlobalList
   repeat with i = 97 to 122
      put filterStackScriptForGlobals(numtochar(i)) & cr after  gGlobalList
   end repeat
   filter lines of gGlobalList without empty
   set the customPropertySet of this card to empty
   put the global_prefix_ignore_list of this card into gGlobalPrefixIgnoreList
   if gGlobalPrefixIgnoreList is not empty then
      repeat for each line tIgnore in gGlobalPrefixIgnoreList
         filter lines of gGlobalList without tIgnore& "*"
      end repeat
   end if
   
   sort gGlobalList
   repeat for each line L in gGlobalList
      do "put" && L &&"into tVar"
      try
         do "put L &&" && quoted(tVar)  &&"&& cr after tAllCurrentGlobalValues"
      catch err
         put err & cr after tAllCurrentGlobalValues
      finally
         do "put L &&" && tVar  &&"& cr after tAllCurrentGlobalValues"
      end try
   end repeat
   put tAllCurrentGlobalValues
end getAllCurrentGlobalValues
